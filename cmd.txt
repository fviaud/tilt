# Installation of kind
curl -Lo ./kind https://kind.sigs.k8s.io/dl/v0.30.0/kind-linux-amd64 && chmod +x ./kind && sudo mv ./kind /usr/local/bin/kind

# installation of tilt
curl -fsSL https://raw.githubusercontent.com/tilt-dev/tilt/master/scripts/install.sh | bash

# Installation of kubectl
curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
chmod +x kubectl
sudo mv kubectl /usr/local/bin/

# Installation of Docker registry
docker run -d -p 5000:5000 --restart=always --name kind-registry registry:2

# Create a cluster with the local registry enabled in containerd
kind create cluster --config kind-config.yaml

# Installation of nginx ingress controller
kubectl apply -f https://raw.githubusercontent.com/kubernetes/ingress-nginx/main/deploy/static/provider/kind/deploy.yaml 2>&1

# Exemple of deployment
docker build -t localhost:5000/mon-app:v1 frontend/ && docker push localhost:5000/mon-app:v1
kubectl create namespace atelier && kubectl apply -f deployment.yml

# Installation of openssh-server
sudo apt update
sudo apt install openssh-server
sudo service ssh start

# Configuration of WSL to avoid the automatic generation of /etc/hosts
sudo nano /etc/wsl.conf
[network]
generateHosts = false

sudo nano /etc/hosts
127.0.0.1 frontend.local
127.0.0.1 keycloak.local

# Cleanup of Docker images, containers and volumes
docker system prune -a --volumes -f
